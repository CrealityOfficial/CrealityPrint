cmake_minimum_required(VERSION 3.15)
project(device_add_tests LANGUAGES CXX)
enable_testing()
add_executable(device_add_tests device_add_tests.cpp)
target_compile_features(device_add_tests PRIVATE cxx_std_17)
target_include_directories(device_add_tests PRIVATE ../../src)
if(MSVC)
    target_compile_options(device_add_tests PRIVATE /utf-8)
endif()
add_test(NAME device_add_tests COMMAND device_add_tests)
