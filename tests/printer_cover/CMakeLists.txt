cmake_minimum_required(VERSION 3.15)
project(printer_cover_tests LANGUAGES CXX)
enable_testing()

set(Boost_USE_STATIC_LIBS ON)
find_package(Boost REQUIRED COMPONENTS log thread filesystem chrono atomic date_time)
set(wxWidgets_USE_STATIC ON)
find_package(wxWidgets REQUIRED COMPONENTS base core)
include(${wxWidgets_USE_FILE})
# Our static wxWidgets dependency uses external image codecs.
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(TIFF REQUIRED)
find_package(ZLIB REQUIRED)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../cmake/modules")
find_package(FFmpeg QUIET)

set(SLIC3R_GENERATED_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../build/src/libslic3r"
    CACHE PATH "Directory containing generated libslic3r_version.h")
add_executable(printer_cover_tests printer_cover_tests.cpp ../../src/slic3r/Utils/PrinterCover.cpp)
target_compile_features(printer_cover_tests PRIVATE cxx_std_17)
target_compile_definitions(printer_cover_tests PRIVATE BOOST_ALL_NO_LIB)
target_include_directories(printer_cover_tests PRIVATE ../../src ${SLIC3R_GENERATED_INCLUDE_DIR} ${Boost_INCLUDE_DIRS})
target_link_libraries(printer_cover_tests PRIVATE ${wxWidgets_LIBRARIES} ${Boost_LIBRARIES} PNG::PNG JPEG::JPEG TIFF::TIFF ZLIB::ZLIB)
if(FFMPEG_FOUND)
    target_compile_definitions(printer_cover_tests PRIVATE ENABLE_FFMPEG=1)
    target_link_libraries(printer_cover_tests PRIVATE FFmpeg::avcodec FFmpeg::avutil FFmpeg::swscale)
endif()
if(MSVC)
    target_compile_options(printer_cover_tests PRIVATE /utf-8)
endif()
add_test(NAME printer_cover_tests COMMAND printer_cover_tests)
