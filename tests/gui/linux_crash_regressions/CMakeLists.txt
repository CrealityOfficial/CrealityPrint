cmake_minimum_required(VERSION 3.16)
project(LinuxCrashRegressions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(wxWidgets REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})
find_package(PkgConfig REQUIRED)
pkg_check_modules(PANGO REQUIRED IMPORTED_TARGET pangocairo pangoft2)
set(PANGO_RUNTIME_DIR "" CACHE PATH "Private Pango runtime lib directory from dep_Pango")
if (NOT EXISTS "${PANGO_RUNTIME_DIR}/libpango-1.0.so.0")
    message(FATAL_ERROR "Set PANGO_RUNTIME_DIR to the built private runtime's lib directory")
endif ()
add_executable(local_drag_tests local_drag_tests.cpp)
# Match the application's wxWidgets build, which omits wx assertion symbols.
target_compile_definitions(local_drag_tests PRIVATE wxDEBUG_LEVEL=0)
target_link_libraries(local_drag_tests PRIVATE ${wxWidgets_LIBRARIES})
add_executable(event_source_lifetime_tests event_source_lifetime_tests.cpp)
target_compile_definitions(event_source_lifetime_tests PRIVATE wxDEBUG_LEVEL=0)
target_link_libraries(event_source_lifetime_tests PRIVATE ${wxWidgets_LIBRARIES})
add_test(NAME event_source_lifetime COMMAND event_source_lifetime_tests)
add_executable(pango_fallback_tests pango_fallback_tests.c)
target_link_libraries(pango_fallback_tests PRIVATE
    "${PANGO_RUNTIME_DIR}/libpangocairo-1.0.so.0"
    "${PANGO_RUNTIME_DIR}/libpangoft2-1.0.so.0"
    "${PANGO_RUNTIME_DIR}/libpango-1.0.so.0" PkgConfig::PANGO)
set_target_properties(pango_fallback_tests PROPERTIES BUILD_RPATH "${PANGO_RUNTIME_DIR}")
add_test(NAME pango_fallback COMMAND pango_fallback_tests)
set_tests_properties(pango_fallback PROPERTIES ENVIRONMENT "G_DEBUG=fatal-criticals")
find_program(XVFB_RUN xvfb-run REQUIRED)
add_test(NAME local_drag COMMAND ${XVFB_RUN} -a $<TARGET_FILE:local_drag_tests>)
set_tests_properties(local_drag PROPERTIES ENVIRONMENT "GDK_BACKEND=x11")
