cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CloudDeviceMqttTests LANGUAGES CXX)
    enable_testing()
endif()
find_package(Threads REQUIRED)
set(CLOUD_MQTT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../src/slic3r/GUI/print_manage/App")
add_executable(cloud_device_mqtt_session_test
    cloud_device_mqtt_session_test.cpp
    "${CLOUD_MQTT_SOURCE_DIR}/CloudDeviceMqttSession.cpp")
target_include_directories(cloud_device_mqtt_session_test PRIVATE "${CLOUD_MQTT_SOURCE_DIR}")
target_compile_features(cloud_device_mqtt_session_test PRIVATE cxx_std_17)
target_link_libraries(cloud_device_mqtt_session_test PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(cloud_device_mqtt_session_test PRIVATE /UNDEBUG /utf-8)
else()
    target_compile_options(cloud_device_mqtt_session_test PRIVATE -UNDEBUG)
endif()
add_test(NAME cloud_device_mqtt_session COMMAND cloud_device_mqtt_session_test)
